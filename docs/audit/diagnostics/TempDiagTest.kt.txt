package com.roundtrace.app.feature

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import androidx.compose.ui.test.junit4.createComposeRule
import androidx.compose.ui.test.onNodeWithText
import androidx.compose.ui.test.onRoot

import androidx.compose.ui.test.printToString
import com.roundtrace.app.core.ui.theme.RoundTraceTheme
import com.roundtrace.app.feature.record.ProposalSheet
import org.junit.Rule
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config

@RunWith(RobolectricTestRunner::class)
@Config(sdk = [34], qualifiers = "zh-rCN-w411dp-h891dp")
class TempDiagTest {
    @get:Rule
    val compose = createComposeRule()

    @Test
    fun dumpTree() {
        var leader by mutableStateOf<Int?>(1)
        var team by mutableStateOf(setOf(1, 2, 3))
        compose.setContent {
            RoundTraceTheme {
                ProposalSheet(
                    questNo = 1,
                    expectedTeamSize = 3,
                    playerCount = 8,
                    leaderSeat = leader,
                    teamSeats = team,
                    canSubmit = leader != null && team.size == 3,
                    isEditing = true,
                    onSelectLeader = { leader = it },
                    onToggleTeamSeat = { s -> team = if (s in team) team - s else team + s },
                    onSaveDraft = {},
                    onSubmit = {},
                    onDismiss = {}
                )
            }
        }
        compose.waitForIdle()
        val roots = compose.onAllNodes(androidx.compose.ui.test.isRoot())
        val sb = StringBuilder("ROOT_COUNT=${roots.fetchSemanticsNodes().size}\n")
        sb.append("ROOT#1: ").append(roots[1].printToString(maxDepth = 100)).append("\n")
        throw AssertionError("TREE>>>\n$sb\n<<<TREE")
    }
}
